## ICellInputSource.cs

```csharp
public interface ICellInputSource
{
    event Action<int> CellSelected;   // ← „pacientul a selectat căsuța i” (0..8)
    bool InputEnabled { get; set; }   // ← controller-ul o pornește doar pe rândul pacientului
}
```
Orice clasă care poate „alege o căsuță” implementează această interfață: acum `BoardView` (click),
mai târziu în VR o clasă care detectează atingerea mâinii. `GameController` nu vede diferența.

Exemplu de cum ar arăta în VR (doar ca idee, nu e în proiect):
```csharp
public class HandTouchCellInput : MonoBehaviour, ICellInputSource
{
    public event Action<int> CellSelected;
    public bool InputEnabled { get; set; }
    // apelată de un collider de pe căsuța `cell` când îl atinge degetul:
    public void OnFingerTouched(int cell) { if (InputEnabled) CellSelected?.Invoke(cell); }
}
```

### `Circle` – un cerc desenat pixel cu pixel
```csharp
public static Sprite Circle
{
    get
    {
        if (circle != null) return circle;     // ← creat deja? îl refolosim
        const int size = 256;
        var tex = new Texture2D(size, size, TextureFormat.RGBA32, false) { wrapMode = TextureWrapMode.Clamp };
        // ← textură goală 256x256, format RGBA, fără mipmap-uri (false).
        //   `{ wrapMode = ... }` = "object initializer": setează proprietăți imediat după construcție
        float r = size / 2f;                   // ← raza = 128
        for (int y = 0; y < size; y++)
        for (int x = 0; x < size; x++)         // ← parcurgem toți pixelii
        {
            float d = Vector2.Distance(new Vector2(x + 0.5f, y + 0.5f), new Vector2(r, r));
            // ← distanța de la centrul pixelului la centrul imaginii
            tex.SetPixel(x, y, new Color(1, 1, 1, Mathf.Clamp01(r - d)));
            // ← alb; alpha = 1 în interior, 0 în exterior, iar pe margine (r-d între 0 și 1) o tranziție fină → margine netedă
        }
        tex.Apply();                           // ← trimite pixelii la placa video
        circle = Sprite.Create(tex, new Rect(0, 0, size, size), new Vector2(0.5f, 0.5f));
        // ← Sprite = textură + dreptunghiul folosit + pivotul (centru)
        return circle;
    }
}
```
Așa nu avem nevoie de niciun fișier imagine pentru monedă.

### Poziționare: `Place`, `Centered`, `Stretch`
```csharp
public static RectTransform Place(this RectTransform rt, Vector2 anchor, Vector2 position, Vector2 size)
{
    rt.anchorMin = rt.anchorMax = anchor;      // ← ambele ancore în același punct (ex. 0.5,0.5 = centrul părintelui)
    rt.pivot = new Vector2(0.5f, 0.5f);        // ← poziționăm după centrul elementului
    rt.anchoredPosition = position;            // ← deplasarea față de ancoră (ex. (0, 400) = 400 px mai sus)
    rt.sizeDelta = size;                       // ← lățimea și înălțimea
    return rt;                                 // ← returnăm rt ca să putem înlănțui apeluri
}

public static RectTransform Centered(this RectTransform rt, Vector2 position, Vector2 size) =>
    rt.Place(new Vector2(0.5f, 0.5f), position, size);   // ← scurtătură: ancorat în centru

public static RectTransform Stretch(this RectTransform rt)
{
    rt.anchorMin = Vector2.zero;               // ← colțul stânga-jos al părintelui
    rt.anchorMax = Vector2.one;                // ← colțul dreapta-sus al părintelui
    rt.offsetMin = rt.offsetMax = Vector2.zero;// ← fără margini → ocupă tot părintele
    return rt;
}
```
Sunt **metode de extensie** (`this RectTransform rt`), deci scrii `label.rectTransform.Centered(...)`.

### `Panel`, `Label`, `Button`
```csharp
public static Image Panel(string name, Transform parent, Color color)
{
    var image = Rect(name, parent).gameObject.AddComponent<Image>();  // ← obiect UI + componenta Image
    image.color = color;                       // ← fără sprite, Image desenează un dreptunghi plin
    return image;
}

public static Text Label(string name, Transform parent, string text, int fontSize, Color color,
    FontStyle style = FontStyle.Normal)        // ← parametru cu valoare implicită (ca în C++)
{
    var label = Rect(name, parent).gameObject.AddComponent<Text>();
    label.font = Font;
    label.text = text;
    label.fontSize = fontSize;
    label.fontStyle = style;
    label.color = color;
    label.alignment = TextAnchor.MiddleCenter;               // ← centrat orizontal și vertical
    label.horizontalOverflow = HorizontalWrapMode.Wrap;      // ← textul lung trece pe rândul următor
    label.verticalOverflow = VerticalWrapMode.Overflow;      // ← poate depăși înălțimea fără să fie tăiat
    label.raycastTarget = false;               // ← textul NU interceptează click-uri (le lasă butonului de sub el)
    return label;
}

public static Button Button(string name, Transform parent, string text, int fontSize, Color background,
    Color textColor, UnityAction onClick)      // ← UnityAction = varianta Unity a lui Action
{
    var image = Panel(name, parent, background);             // ← fundalul butonului
    var button = image.gameObject.AddComponent<Button>();    // ← logica de click
    button.targetGraphic = image;              // ← imaginea care își schimbă culoarea la hover/apăsare

    var colors = ColorBlock.defaultColorBlock; // ← culorile implicite ale stărilor
    colors.highlightedColor = new Color(0.92f, 0.92f, 0.92f);  // ← la hover: puțin mai închis
    colors.pressedColor = new Color(0.78f, 0.78f, 0.78f);      // ← la apăsare: și mai închis
    colors.selectedColor = Color.white;        // ← după click rămâne normal (nu „selectat”)
    button.colors = colors;
    // ← culorile acestea se ÎNMULȚESC cu culoarea imaginii (alb = neschimbat)

    var label = Label("Label", image.transform, text, fontSize, textColor, FontStyle.Bold);  // ← textul, copil al butonului
    label.rectTransform.Stretch();             // ← ocupă tot butonul

    if (onClick != null) button.onClick.AddListener(onClick);  // ← funcția apelată la click
    return button;
}

public static Text LabelOf(Button button) => button.GetComponentInChildren<Text>();
// ← găsește componenta Text din copiii butonului (ca să-i putem schimba textul ulterior)
```

### `Build` – cele 9 căsuțe
```csharp
    void Build(float size)
    {
        float gap = size * 0.025f;             // ← spațiul dintre căsuțe (2.5% din tablă)
        float cell = (size - gap * 4f) / Board.Size;
        // ← latura unei căsuțe: din lățime scădem 4 spații (margine, 2 între căsuțe, margine) și împărțim la 3

        for (int i = 0; i < Board.CellCount; i++)
        {
            int index = i;
            // ← COPIE a lui i. Lambda de mai jos capturează variabila; fără copie, toate butoanele
            //   ar vedea valoarea finală a lui i (9). Capcană clasică în C#.
            float x = -size / 2f + gap + cell / 2f + Board.Column(i) * (cell + gap);
            // ← centrul căsuței pe X: pornim din marginea stângă (-size/2), sărim marginea și o jumătate de căsuță,
            //   apoi câte (căsuță + spațiu) pentru fiecare coloană
            float y = size / 2f - gap - cell / 2f - Board.Row(i) * (cell + gap);
            // ← la fel pe Y, dar de sus în jos (în UI Unity, Y crește în SUS)

            var button = UIFactory.Button($"Cell_{i}", transform, "", Mathf.RoundToInt(cell * 0.75f),
                Theme.Cell, Theme.Text, () => OnCellClicked(index));
            // ← buton alb, fără text, cu font de 75% din căsuță; la click → OnCellClicked(index)
            ((RectTransform)button.transform).Centered(new Vector2(x, y), new Vector2(cell, cell));

            cellImages[i] = (Image)button.targetGraphic;   // ← memorăm imaginea (pentru colorarea liniei câștigătoare)
            cellTexts[i] = UIFactory.LabelOf(button);      // ← memorăm textul (pentru X/0)
        }
    }
```
**Truc:** nu desenăm linii. Tabla e un pătrat închis la culoare, iar căsuțele albe au spații între ele –
fundalul care se vede printre ele arată ca o grilă.

### `Pop` – animația
```csharp
    static IEnumerator Pop(Transform target)
    {
        const float duration = 0.25f;          // ← un sfert de secundă
        for (float t = 0; t < duration; t += Time.deltaTime)   // ← t crește cu timpul fiecărui frame
        {
            float k = t / duration;            // ← progres 0 → 1
            float scale = Mathf.LerpUnclamped(0.3f, 1f, 1f + 2.7f * Mathf.Pow(k - 1f, 3) + 1.7f * Mathf.Pow(k - 1f, 2));
            // ← formula "ease-out-back": pornește de la 30%, crește puțin PESTE 100% și revine la 100%
            //   (LerpUnclamped permite depășirea lui 1, de aici efectul de „săritură”)
            target.localScale = new Vector3(scale, scale, 1f);
            yield return null;                 // ← așteaptă frame-ul următor
        }
        target.localScale = Vector3.one;       // ← la final, exact 100%
    }
```

## Screens.cs – cele 4 ecrane

Toate urmează același tipar:
1. `static Create(parent)` creează un panou care ocupă tot ecranul, pune componenta pe el și apelează `Build()`.
2. `Build()` creează elementele (texte, butoane).
3. Comunicarea cu exteriorul se face **doar prin evenimente** – ecranele nu știu de `GameController`.

#### `Play` – animația monedei
```csharp
    public IEnumerator Play(bool patientStarts, float resultHoldSeconds = 1.6f)
    {
        title.text = "Aruncăm moneda…";
        resultLabel.text = "";

        int halfTurns = UnityEngine.Random.Range(7, 10);
        // ← câte întoarceri face moneda: 7, 8 sau 9 (la int, Range exclude capătul de sus)
        bool showingPatient = (halfTurns % 2 == 0) == patientStarts;
        // ← fața de pornire: dacă numărul de întoarceri e par, moneda termină pe aceeași față cu care a început,
        //   deci începem cu rezultatul; dacă e impar, începem cu fața opusă. Astfel aterizează mereu corect.
        SetFace(showingPatient);

        for (int i = 0; i < halfTurns; i++)
        {
            float duration = Mathf.Lerp(0.06f, 0.22f, i / (float)(halfTurns - 1)); // încetinește spre final
            // ← prima întoarcere durează 0.06 s, ultima 0.22 s. (float) ca împărțirea să nu fie întreagă
            yield return Scale(1f, 0f, duration);   // ← moneda se „îngustează” până devine o linie (pe muchie)
            showingPatient = !showingPatient;       // ← schimbăm fața când nu se vede
            SetFace(showingPatient);
            yield return Scale(0f, 1f, duration);   // ← se „lărgește” la loc, cu cealaltă față
        }

        title.text = patientStarts ? "Ai câștigat aruncarea!" : "AI-ul a câștigat aruncarea";
        resultLabel.text = patientStarts ? "Începi tu – tu ești X" : "Începe AI-ul – tu ești 0";
        yield return new WaitForSeconds(resultHoldSeconds);   // ← lăsăm rezultatul pe ecran 1.6 s
    }

    void SetFace(bool patient)
    {
        coin.color = patient ? PatientFace : AIFace;
        coinLabel.text = patient ? "TU" : "AI";
    }

    IEnumerator Scale(float from, float to, float duration)
    {
        for (float t = 0; t < duration; t += Time.deltaTime)
        {
            coin.transform.localScale = new Vector3(Mathf.Lerp(from, to, t / duration), 1f, 1f);
            // ← modificăm doar scala pe X → iluzia unei rotații 3D, fără 3D
            yield return null;
        }
        coin.transform.localScale = new Vector3(to, 1f, 1f);   // ← valoarea finală exactă
    }
}
```

> **De ce sunt 4 MonoBehaviour-uri într-un singur fișier?** Unity cere „un MonoBehaviour per fișier, cu același
> nume” doar pentru componentele pe care le tragi manual în Inspector. Componentele adăugate din cod cu
> `AddComponent<T>()` funcționează oricum. Dacă vrei vreodată să le pui manual, le muți în fișiere separate.
